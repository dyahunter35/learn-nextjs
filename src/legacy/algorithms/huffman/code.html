ترميز هوفمان — الكود البرمجي

شرح مبسط لخوارزمية ترميز هوفمان لضغط البيانات مع أمثلة برمجية بلغات متعددة. يمكنك نسخ الكود واستخدامه مباشرة.

كيف تعمل خوارزمية ترميز هوفمان:
ترميز هوفمان هو خوارزمية ضغط بيانات تفقدية تستخدم لتقليل حجم البيانات عن طريق ترميز الأحرف الأكثر تكراراً بأكواد أقصر. تقوم الخوارزمية ببناء شجرة ثنائية حيث تكون الأحرف الأقل تكراراً في المستويات الأعمق (أكواد أطول)، والأحرف الأكثر تكراراً في المستويات الأعلى (أكواد أقصر). يتم استخدام هذه الشجرة لإنشاء أكواد ثنائية فريدة لكل حرف. 
#include <iostream>
#include <string>
#include <queue>
#include <unordered_map>
#include <vector>
using namespace std;

struct Node {
    char ch;
    int freq;
    Node *left, *right;
    Node(char c, int f) : ch(c), freq(f), left(nullptr), right(nullptr) {}
};

struct Compare {
    bool operator()(Node* a, Node* b) {
        return a->freq > b->freq;
    }
};

void generateCodes(Node* root, string code, unordered_map<char, string>& codes) {
    if (!root) return;
    if (!root->left && !root->right) {
        codes[root->ch] = code;
    }
    generateCodes(root->left, code + "0", codes);
    generateCodes(root->right, code + "1", codes);
}

string encode(const string& text, unordered_map<char, string>& codes) {
    string result = "";
    for (char ch : text) result += codes[ch];
    return result;
}

string decode(Node* root, const string& bits) {
    string result = "";
    Node* curr = root;
    for (char b : bits) {
        curr = (b == '0') ? curr->left : curr->right;
        if (!curr->left && !curr->right) {
            result += curr->ch;
            curr = root;
        }
    }
    return result;
}

int main() {
    string text = "aabbbccdeeeeff";

    unordered_map<char, int> freq;
    for (char ch : text) freq[ch]++;

    priority_queue<Node*, vector<Node*>, Compare> pq;
    for (auto& p : freq) pq.push(new Node(p.first, p.second));

    while (pq.size() > 1) {
        Node* left = pq.top(); pq.pop();
        Node* right = pq.top(); pq.pop();
        Node* merged = new Node('\0', left->freq + right->freq);
        merged->left = left;
        merged->right = right;
        pq.push(merged);
    }

    Node* root = pq.top();
    unordered_map<char, string> codes;
    generateCodes(root, "", codes);

    cout << "Huffman Codes:" << endl;
    for (auto& p : codes) {
        cout << p.first << ": " << p.second << endl;
    }

    string encoded = encode(text, codes);
    string decoded = decode(root, encoded);

    cout << "\nOriginal: " << text << endl;
    cout << "Encoded: " << encoded << endl;
    cout << "Decoded: " << decoded << endl;

    return 0;
}
import java.util.*;

class HuffmanNode implements Comparable<HuffmanNode> {
    char ch;
    int freq;
    HuffmanNode left, right;

    HuffmanNode(char ch, int freq) {
        this.ch = ch;
        this.freq = freq;
    }

    public int compareTo(HuffmanNode other) {
        return this.freq - other.freq;
    }
}

public class HuffmanCoding {

    public static void generateCodes(HuffmanNode root, String code, Map<Character, String> codes) {
        if (root == null) return;
        if (root.left == null && root.right == null) {
            codes.put(root.ch, code);
        }
        generateCodes(root.left, code + "0", codes);
        generateCodes(root.right, code + "1", codes);
    }

    public static String encode(String text, Map<Character, String> codes) {
        StringBuilder sb = new StringBuilder();
        for (char ch : text.toCharArray()) sb.append(codes.get(ch));
        return sb.toString();
    }

    public static String decode(HuffmanNode root, String bits) {
        StringBuilder sb = new StringBuilder();
        HuffmanNode curr = root;
        for (char b : bits.toCharArray()) {
            curr = (b == '0') ? curr.left : curr.right;
            if (curr.left == null && curr.right == null) {
                sb.append(curr.ch);
                curr = root;
            }
        }
        return sb.toString();
    }

    public static void main(String[] args) {
        String text = "aabbbccdeeeeff";

        Map<Character, Integer> freq = new HashMap<>();
        for (char ch : text.toCharArray()) freq.put(ch, freq.getOrDefault(ch, 0) + 1);

        PriorityQueue<HuffmanNode> pq = new PriorityQueue<>();
        for (Map.Entry<Character, Integer> e : freq.entrySet()) {
            pq.offer(new HuffmanNode(e.getKey(), e.getValue()));
        }

        while (pq.size() > 1) {
            HuffmanNode left = pq.poll();
            HuffmanNode right = pq.poll();
            HuffmanNode merged = new HuffmanNode('\0', left.freq + right.freq);
            merged.left = left;
            merged.right = right;
            pq.offer(merged);
        }

        HuffmanNode root = pq.poll();
        Map<Character, String> codes = new HashMap<>();
        generateCodes(root, "", codes);

        System.out.println("Huffman Codes:");
        for (Map.Entry<Character, String> e : codes.entrySet()) {
            System.out.println(e.getKey() + ": " + e.getValue());
        }

        String encoded = encode(text, codes);
        String decoded = decode(root, encoded);

        System.out.println("\nOriginal: " + text);
        System.out.println("Encoded: " + encoded);
        System.out.println("Decoded: " + decoded);
    }
}
class Node {
    constructor(ch, freq) {
        this.ch = ch;
        this.freq = freq;
        this.left = null;
        this.right = null;
    }
}

function generateCodes(root, code, codes) {
    if (!root) return;
    if (!root.left && !root.right) {
        codes[root.ch] = code;
    }
    generateCodes(root.left, code + "0", codes);
    generateCodes(root.right, code + "1", codes);
}

function encode(text, codes) {
    return text.split("").map(ch => codes[ch]).join("");
}

function decode(root, bits) {
    let result = "";
    let curr = root;
    for (const b of bits) {
        curr = b === "0" ? curr.left : curr.right;
        if (!curr.left && !curr.right) {
            result += curr.ch;
            curr = root;
        }
    }
    return result;
}

function buildHuffmanTree(text) {
    const freq = {};
    for (const ch of text) freq[ch] = (freq[ch] || 0) + 1;

    const pq = Object.entries(freq).map(([ch, f]) => new Node(ch, f));
    pq.sort((a, b) => a.freq - b.freq);

    while (pq.length > 1) {
        const left = pq.shift();
        const right = pq.shift();
        const merged = new Node("", left.freq + right.freq);
        merged.left = left;
        merged.right = right;
        pq.push(merged);
        pq.sort((a, b) => a.freq - b.freq);
    }

    return pq[0];
}

<?php

class Node {
    public $ch;
    public $freq;
    public $left;
    public $right;

    public function __construct($ch, $freq) {
        $this->ch = $ch;
        $this->freq = $freq;
        $this->left = null;
        $this->right = null;
    }
}

function generateCodes($root, $code, &$codes) {
    if ($root === null) return;
    if ($root->left === null && $root->right === null) {
        $codes[$root->ch] = $code;
    }
    generateCodes($root->left, $code . "0", $codes);
    generateCodes($root->right, $code . "1", $codes);
}

function encodeText($text, $codes) {
    $result = "";
    for ($i = 0; $i < strlen($text); $i++) {
        $result .= $codes[$text[$i]];
    }
    return $result;
}

function decodeText($root, $bits) {
    $result = "";
    $curr = $root;
    for ($i = 0; $i < strlen($bits); $i++) {
        $curr = $bits[$i] === "0" ? $curr->left : $curr->right;
        if ($curr->left === null && $curr->right === null) {
            $result .= $curr->ch;
            $curr = $root;
        }
    }
    return $result;
}

function buildTree($text) {
    $freq = [];
    for ($i = 0; $i < strlen($text); $i++) {
        $ch = $text[$i];
        if (!isset($freq[$ch])) $freq[$ch] = 0;
        $freq[$ch]++;
    }

    $pq = [];
    foreach ($freq as $ch => $f) {
        $pq[] = new Node($ch, $f);
    }

    while (count($pq) > 1) {
        usort($pq, function($a, $b) { return $a->freq - $b->freq; });
        $left = array_shift($pq);
        $right = array_shift($pq);
        $merged = new Node("", $left->freq + $right->freq);
        $merged->left = $left;
        $merged->right = $right;
        $pq[] = $merged;
    }

    return $pq[0];
}

$text = "aabbbccdeeeeff";
$root = buildTree($text);
$codes = [];
generateCodes($root, "", $codes);

echo "Huffman Codes:\n";
foreach ($codes as $ch => $code) {
    echo "$ch: $code\n";
}

$encoded = encodeText($text, $codes);
$decoded = decodeText($root, $encoded);

echo "\nOriginal: $text\n";
echo "Encoded: $encoded\n";
echo "Decoded: $decoded\n";
