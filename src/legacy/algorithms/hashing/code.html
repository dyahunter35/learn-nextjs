التجزئة (Hashing) — الكود البرمجي

شرح مبسط مع أمثلة بلغات متعددة. يمكنك نسخ الكود واستخدامه.

كيف تعمل: جدول التجزئة (Hash Table) يخزن البيانات باستخدام دالة تجزئة تحول المفتاح إلى فهرس في مصفوفة. تستخدم هذه الخوارزمية التسلسل (Chaining) لحل التصادمات — كل خلية تحتوي على قائمة مرتبطة (linked list) من العناصر التي تشترك في نفس قيمة التجزئة. دالة التجزئة: key % size. 
#include <iostream>
#include <list>
#include <vector>
using namespace std;

class HashTable {
    int size;
    vector<list<int>> table;

    int hash(int key) {
        return key % size;
    }

public:
    HashTable(int s) : size(s) {
        table.resize(size);
    }

    void insert(int key) {
        int idx = hash(key);
        table[idx].push_back(key);
    }

    bool search(int key) {
        int idx = hash(key);
        for (int x : table[idx])
            if (x == key) return true;
        return false;
    }

    void remove(int key) {
        int idx = hash(key);
        table[idx].remove(key);
    }

    void print() {
        for (int i = 0; i < size; i++) {
            cout << "[" << i << "]: ";
            for (int x : table[i])
                cout << x << " -> ";
            cout << "null" << endl;
        }
    }
};

int main() {
    HashTable ht(7);
    ht.insert(10);
    ht.insert(20);
    ht.insert(15);
    ht.insert(7);

    cout << "Hash table (size=7):" << endl;
    ht.print();

    int key = 15;
    cout << "Search " << key << ": "
         << (ht.search(key) ? "Found" : "Not found") << endl;

    ht.remove(15);
    cout << "After removing 15, search: "
         << (ht.search(15) ? "Found" : "Not found") << endl;
    return 0;
}
import java.util.LinkedList;

class HashTable {
    private int size;
    private LinkedList<Integer>[] table;

    @SuppressWarnings("unchecked")
    public HashTable(int s) {
        size = s;
        table = new LinkedList[size];
        for (int i = 0; i < size; i++)
            table[i] = new LinkedList<>();
    }

    private int hash(int key) {
        return key % size;
    }

    public void insert(int key) {
        int idx = hash(key);
        table[idx].add(key);
    }

    public boolean search(int key) {
        int idx = hash(key);
        return table[idx].contains(key);
    }

    public void remove(int key) {
        int idx = hash(key);
        table[idx].remove(Integer.valueOf(key));
    }

    public void print() {
        for (int i = 0; i < size; i++) {
            System.out.print("[" + i + "]: ");
            for (int x : table[i])
                System.out.print(x + " -> ");
            System.out.println("null");
        }
    }
}

class HashTable {
    constructor(size) {
        this.size = size;
        this.table = new Array(size).fill(null).map(() => []);
    }

    hash(key) {
        return key % this.size;
    }

    insert(key) {
        const idx = this.hash(key);
        this.table[idx].push(key);
    }

    search(key) {
        const idx = this.hash(key);
        return this.table[idx].includes(key);
    }

    remove(key) {
        const idx = this.hash(key);
        this.table[idx] = this.table[idx].filter(x => x !== key);
    }

    print() {
        for (let i = 0; i < this.size; i++)
            console.log(`[${i}]: ${this.table[i].join(" -> ") || "null"}`);
    }
}

<?php

class HashTable {
    private int $size;
    private array $table;

    public function __construct(int $size) {
        $this->size = $size;
        $this->table = array_fill(0, $size, []);
    }

    private function hash(int $key): int {
        return $key % $this->size;
    }

    public function insert(int $key): void {
        $idx = $this->hash($key);
        $this->table[$idx][] = $key;
    }

    public function search(int $key): bool {
        $idx = $this->hash($key);
        return in_array($key, $this->table[$idx]);
    }

    public function remove(int $key): void {
        $idx = $this->hash($key);
        $this->table[$idx] = array_values(
            array_filter($this->table[$idx], fn($x) => $x !== $key)
        );
    }

    public function print(): void {
        for ($i = 0; $i < $this->size; $i++) {
            $chain = empty($this->table[$i]) ? "null"
                : implode(" -> ", $this->table[$i]);
            echo "[$i]: $chain\n";
        }
    }
}

$ht = new HashTable(7);
$ht->insert(10);
$ht->insert(20);
$ht->insert(15);
$ht->insert(7);

echo "Hash table (size=7):\n";
$ht->print();

$key = 15;
echo "Search $key: " . ($ht->search($key) ? "Found" : "Not found") . "\n";

$ht->remove(15);
echo "After removing 15, search: " . ($ht->search(15) ? "Found" : "Not found") . "\n";

?>
