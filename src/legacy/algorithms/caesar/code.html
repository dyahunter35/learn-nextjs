تشفير قيصر — الكود البرمجي

شرح مبسط لخوارزمية تشفير قيصر مع أمثلة برمجية بلغات متعددة. يمكنك نسخ الكود واستخدامه مباشرة.

كيف تعمل خوارزمية تشفير قيصر:
تعتمد خوارزمية تشفير قيصر على إزاحة كل حرف في النص الأصلي بعدد ثابت من الخطوات في الأبجدية. مثلاً، مع إزاحة مقدارها 3، يصبح الحرف "A" هو "D". لفك التشفير، يتم إرجاع الحرف بنفس عدد الخطوات إلى الخلف. تعتبر هذه الخوارزمية من أبسط طرق التشفير وأكثرها شهرة منذ العصور الرومانية. 
#include <iostream>
#include <string>
using namespace std;

string encrypt(string text, int shift) {
    string result = "";
    for (char ch : text) {
        if (isupper(ch)) {
            result += char((ch - 'A' + shift) % 26 + 'A');
        } else if (islower(ch)) {
            result += char((ch - 'a' + shift) % 26 + 'a');
        } else {
            result += ch;
        }
    }
    return result;
}

string decrypt(string text, int shift) {
    return encrypt(text, 26 - shift);
}

int main() {
    string text = "Hello Caesar Cipher!";
    int shift = 3;

    string encrypted = encrypt(text, shift);
    string decrypted = decrypt(encrypted, shift);

    cout << "Original: " << text << endl;
    cout << "Encrypted (shift=" << shift << "): " << encrypted << endl;
    cout << "Decrypted: " << decrypted << endl;

    return 0;
}
public class CaesarCipher {

    public static String encrypt(String text, int shift) {
        StringBuilder result = new StringBuilder();
        for (char ch : text.toCharArray()) {
            if (Character.isUpperCase(ch)) {
                result.append((char) ((ch - 'A' + shift) % 26 + 'A'));
            } else if (Character.isLowerCase(ch)) {
                result.append((char) ((ch - 'a' + shift) % 26 + 'a'));
            } else {
                result.append(ch);
            }
        }
        return result.toString();
    }

    public static String decrypt(String text, int shift) {
        return encrypt(text, 26 - shift);
    }

    public static void main(String[] args) {
        String text = "Hello Caesar Cipher!";
        int shift = 3;

        String encrypted = encrypt(text, shift);
        String decrypted = decrypt(encrypted, shift);

        System.out.println("Original: " + text);
        System.out.println("Encrypted (shift=" + shift + "): " + encrypted);
        System.out.println("Decrypted: " + decrypted);
    }
}
function encrypt(text, shift) {
    let result = "";
    for (const ch of text) {
        if (ch >= 'A' && ch <= 'Z') {
            result += String.fromCharCode((ch.charCodeAt(0) - 65 + shift) % 26 + 65);
        } else if (ch >= 'a' && ch <= 'z') {
            result += String.fromCharCode((ch.charCodeAt(0) - 97 + shift) % 26 + 97);
        } else {
            result += ch;
        }
    }
    return result;
}

function decrypt(text, shift) {
    return encrypt(text, 26 - shift);
}

<?php

function encrypt($text, $shift) {
    $result = "";
    for ($i = 0; $i < strlen($text); $i++) {
        $ch = $text[$i];
        if (ctype_upper($ch)) {
            $result .= chr((ord($ch) - ord('A') + $shift) % 26 + ord('A'));
        } else if (ctype_lower($ch)) {
            $result .= chr((ord($ch) - ord('a') + $shift) % 26 + ord('a'));
        } else {
            $result .= $ch;
        }
    }
    return $result;
}

function decrypt($text, $shift) {
    return encrypt($text, 26 - $shift);
}
