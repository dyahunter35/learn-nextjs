الشجرة الممتدة الدنيا (MST) — الكود البرمجي

خوارزميتا Kruskal و Prim لإيجاد أقل تكلفة لربط جميع نقاط الشبكة مع أمثلة بلغات متعددة.

كيف تعمل Kruskal: ترتيب الحواف حسب الوزن، ثم إضافة كل حافة لا تشكل حلقة باستخدام Union-Find.
كيف تعمل Prim: تبدأ من عقدة وتضيف الحافة الأقل وزناً التي تصل العقدة الممتدة بعقدة جديدة. 
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

struct Edge {
    int src, dest, weight;
};

struct Graph {
    int V, E;
    vector<Edge> edges;
};

struct Subset {
    int parent, rank;
};

int find(Subset subsets[], int i) {
    if (subsets[i].parent != i)
        subsets[i].parent = find(subsets, subsets[i].parent);
    return subsets[i].parent;
}

void unite(Subset subsets[], int x, int y) {
    int rx = find(subsets, x), ry = find(subsets, y);
    if (subsets[rx].rank < subsets[ry].rank)
        subsets[rx].parent = ry;
    else if (subsets[rx].rank > subsets[ry].rank)
        subsets[ry].parent = rx;
    else {
        subsets[ry].parent = rx;
        subsets[rx].rank++;
    }
}

bool cmp(Edge a, Edge b) { return a.weight < b.weight; }

void kruskalMST(Graph& graph) {
    int V = graph.V;
    sort(graph.edges.begin(), graph.edges.end(), cmp);
    Subset* subsets = new Subset[V];
    for (int v = 0; v < V; v++) {
        subsets[v].parent = v;
        subsets[v].rank = 0;
    }
    vector<Edge> result;
    for (Edge e : graph.edges) {
        int x = find(subsets, e.src);
        int y = find(subsets, e.dest);
        if (x != y) {
            result.push_back(e);
            unite(subsets, x, y);
        }
    }
    cout << "MST edges:\n";
    for (Edge e : result)
        cout << e.src << " - " << e.dest << " : " << e.weight << "\n";
    delete[] subsets;
}

class UnionFind {
    constructor(n) {
        this.parent = Array.from({length: n}, (_, i) => i);
        this.rank = new Array(n).fill(0);
    }
    find(i) {
        if (this.parent[i] !== i)
            this.parent[i] = this.find(this.parent[i]);
        return this.parent[i];
    }
    unite(x, y) {
        const rx = this.find(x), ry = this.find(y);
        if (rx === ry) return;
        if (this.rank[rx] < this.rank[ry])
            this.parent[rx] = ry;
        else if (this.rank[rx] > this.rank[ry])
            this.parent[ry] = rx;
        else {
            this.parent[ry] = rx;
            this.rank[rx]++;
        }
    }
}

function kruskalMST(edges, V) {
    edges.sort((a, b) => a.weight - b.weight);
    const uf = new UnionFind(V);
    const result = [];
    for (const e of edges) {
        if (uf.find(e.src) !== uf.find(e.dest)) {
            result.push(e);
            uf.unite(e.src, e.dest);
        }
    }
    return result;
}
