#include <iostream>
#include <vector>
#include <queue>
#include <climits>
using namespace std;

class Graph {
    int V;
    vector<vector<pair<int,int>>> adj;
public:
    Graph(int v) : V(v), adj(v) {}
    void addEdge(int u, int v, int w) {
        adj[u].push_back({v, w});
        adj[v].push_back({u, w});
    }
    void dijkstra(int start) {
        vector<int> dist(V, INT_MAX);
        vector<int> parent(V, -1);
        priority_queue<pair<int,int>, vector<pair<int,int>>, greater<>> pq;
        dist[start] = 0;
        pq.push({0, start});
        while (!pq.empty()) {
            int u = pq.top().second;
            pq.pop();
            for (auto& [v, w] : adj[u]) {
                if (dist[u] + w < dist[v]) {
                    dist[v] = dist[u] + w;
                    parent[v] = u;
                    pq.push({dist[v], v});
                }
            }
        }
        cout << "Distances from " << (char)('A' + start) << ":\n";
        for (int i = 0; i < V; i++) {
            cout << (char)('A' + i) << ": " << dist[i] << "  Path: ";
            vector<int> path;
            for (int v = i; v != -1; v = parent[v])
                path.push_back(v);
            for (int j = path.size()-1; j >= 0; j--)
                cout << (char)('A' + path[j]) << (j ? " -> " : "");
            cout << endl;
        }
    }
};

import java.util.*;

public class Dijkstra {
    static class Edge {
        int dest, weight;
        Edge(int d, int w) { dest = d; weight = w; }
    }

    static class Graph {
        int V;
        List<List<Edge>> adj;
        Graph(int v) {
            V = v;
            adj = new ArrayList<>();
            for (int i = 0; i < v; i++) adj.add(new ArrayList<>());
        }
        void addEdge(int u, int v, int w) {
            adj.get(u).add(new Edge(v, w));
            adj.get(v).add(new Edge(u, w));
        }
        void dijkstra(int start) {
            int[] dist = new int[V];
            int[] parent = new int[V];
            Arrays.fill(dist, Integer.MAX_VALUE);
            Arrays.fill(parent, -1);
            PriorityQueue<int[]> pq = new PriorityQueue<>((a,b) -> a[1] - b[1]);
            dist[start] = 0;
            pq.offer(new int[]{start, 0});
            while (!pq.isEmpty()) {
                int u = pq.poll()[0];
                for (Edge e : adj.get(u)) {
                    if (dist[u] + e.weight < dist[e.dest]) {
                        dist[e.dest] = dist[u] + e.weight;
                        parent[e.dest] = u;
                        pq.offer(new int[]{e.dest, dist[e.dest]});
                    }
                }
            }
            System.out.println("Distances from " + (char)('A' + start) + ":");
            for (int i = 0; i < V; i++) {
                System.out.print((char)('A' + i) + ": " + dist[i] + "  Path: ");
                List<Integer> path = new ArrayList<>();
                for (int v = i; v != -1; v = parent[v]) path.add(v);
                Collections.reverse(path);
                for (int j = 0; j < path.size(); j++) {
                    System.out.print((char)('A' + path.get(j)));
                    if (j < path.size()-1) System.out.print(" -> ");
                }
                System.out.println();
            }
        }
    }

    public static void main(String[] args) {
        Graph g = new Graph(6);
        g.addEdge(0, 1, 7);  // A-B
        g.addEdge(0, 2, 9);  // A-C
        g.addEdge(0, 5, 14); // A-F
        g.addEdge(1, 2, 10); // B-C
        g.addEdge(1, 3, 15); // B-D
        g.addEdge(2, 3, 11); // C-D

        g.addEdge(3, 4, 6);  // D-E
        g.addEdge(4, 5, 9);  // E-F
        g.dijkstra(0);
    }
}
class PriorityQueue {
    constructor() { this.heap = []; }
    enqueue(v, p) {
        this.heap.push({v, p});
        this.heap.sort((a,b) => a.p - b.p);
    }
    dequeue() { return this.heap.shift(); }
    isEmpty() { return this.heap.length === 0; }
}

class Graph {
    constructor(v) {
        this.V = v;
        this.adj = Array.from({length: v}, () => []);
    }
    addEdge(u, v, w) {
        this.adj[u].push({to: v, w});
        this.adj[v].push({to: u, w});
    }
    dijkstra(start) {
        const dist = new Array(this.V).fill(Infinity);
        const parent = new Array(this.V).fill(-1);
        const pq = new PriorityQueue();
        dist[start] = 0;
        pq.enqueue(start, 0);
        while (!pq.isEmpty()) {
            const {v: u} = pq.dequeue();
            for (const {to: v, w} of this.adj[u]) {
                if (dist[u] + w < dist[v]) {
                    dist[v] = dist[u] + w;
                    parent[v] = u;
                    pq.enqueue(v, dist[v]);
                }
            }
        }
        console.log("Distances from " + String.fromCharCode(65 + start) + ":");
        for (let i = 0; i < this.V; i++) {
            let path = [];
            for (let v = i; v !== -1; v = parent[v]) path.push(v);
            path.reverse();
            console.log(String.fromCharCode(65 + i) + ": " + dist[i] +
                "  Path: " + path.map(x => String.fromCharCode(65 + x)).join(" -> "));
        }
    }
}

class Graph {
    private $V;
    private $adj;

    public function __construct($v) {
        $this->V = $v;
        $this->adj = array_fill(0, $v, []);
    }

    public function addEdge($u, $v, $w) {
        $this->adj[$u][] = ['to' => $v, 'w' => $w];
        $this->adj[$v][] = ['to' => $u, 'w' => $w];
    }

    public function dijkstra($start) {
        $dist = array_fill(0, $this->V, PHP_INT_MAX);
        $parent = array_fill(0, $this->V, -1);
        $pq = new SplPriorityQueue();
        $pq->setExtractFlags(SplPriorityQueue::EXTR_DATA);
        $dist[$start] = 0;
        $pq->insert($start, 0);
        while (!$pq->isEmpty()) {
            $u = $pq->extract();
            foreach ($this->adj[$u] as $edge) {
                $v = $edge['to'];
                $w = $edge['w'];
                if ($dist[$u] + $w < $dist[$v]) {
                    $dist[$v] = $dist[$u] + $w;
                    $parent[$v] = $u;
                    $pq->insert($v, -$dist[$v]);
                }
            }
        }
        echo "Distances from " . chr(65 + $start) . ":\n";
        for ($i = 0; $i < $this->V; $i++) {
            $path = [];
            for ($v = $i; $v != -1; $v = $parent[$v]) $path[] = $v;
            $path = array_reverse($path);
            $pathStr = implode(" -> ", array_map(function($x) { return chr(65 + $x); }, $path));
            echo chr(65 + $i) . ": " . $dist[$i] . "  Path: " . $pathStr . "\n";
        }
    }
}
