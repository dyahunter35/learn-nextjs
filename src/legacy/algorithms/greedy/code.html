العملات المعدنية (Greedy Coin Change) — الكود البرمجي

شرح مبسط مع أمثلة بلغات متعددة. يمكنك نسخ الكود واستخدامه.

كيف تعمل: خوارزمية جشعة لمشكلة صرف العملات. تختار دائمًا العملة الأكبر التي لا تتجاوز المبلغ المتبقي، وتكرر حتى يصفر المبلغ. تعمل بكفاءة مع أنظمة العملات القياسية. مثال: عملات سودانية [50, 20, 10, 5, 1] والمبلغ 87. 
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

vector<int> minCoinsGreedy(vector<int> coins, int amount) {
    sort(coins.rbegin(), coins.rend());
    vector<int> result;
    for (int coin : coins) {
        while (amount >= coin) {
            amount -= coin;
            result.push_back(coin);
        }
    }
    return result;
}

int main() {
    vector<int> coins = {50, 20, 10, 5, 1};
    int amount = 87;
    vector<int> result = minCoinsGreedy(coins, amount);
    cout << "Amount: " << amount << endl;
    cout << "Coins used: ";
    for (int c : result) cout << c << " ";
    cout << endl;
    cout << "Total coins: " << result.size() << endl;
    return 0;
}
import java.util.*;

public class CoinChange {
    public static List<Integer> minCoinsGreedy(Integer[] coinsArr, int amount) {
        Arrays.sort(coinsArr, Collections.reverseOrder());
        List<Integer> result = new ArrayList<>();
        for (int coin : coinsArr) {
            while (amount >= coin) {
                amount -= coin;
                result.add(coin);
            }
        }
        return result;
    }

    public static void main(String[] args) {
        Integer[] coins = {50, 20, 10, 5, 1};
        int amount = 87;
        List<Integer> result = minCoinsGreedy(coins, amount);
        System.out.println("Amount: " + amount);
        System.out.print("Coins used: ");
        for (int c : result) System.out.print(c + " ");
        System.out.println();
        System.out.println("Total coins: " + result.size());
    }
}
function minCoinsGreedy(coins, amount) {
    coins.sort((a, b) => b - a);
    const result = [];
    for (const coin of coins) {
        while (amount >= coin) {
            amount -= coin;
            result.push(coin);
        }
    }
    return result;
}
