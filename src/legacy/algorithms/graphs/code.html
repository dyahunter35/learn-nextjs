الرسوم البيانية (BFS/DFS) — الكود البرمجي

شرح مبسط مع أمثلة بلغات متعددة. يمكنك نسخ الكود واستخدامه.

كيف تعمل: تمثل Graph الرسم البياني باستخدام قائمة مجاورة (adjacency list). BFS (البحث بالعرض) يستخدم طابوراً (queue) لزيارة العقدة جاراً بعد جار. DFS (البحث بالعمق) يستخدم مكدساً (stack) أو استدعاء متكرراً للتعمق في المسار قبل العودة. المثال يستخدم 6 عقد A-F. 
#include <iostream>
#include <vector>
#include <queue>
#include <stack>
#include <map>
#include <set>
using namespace std;

class Graph {
    map<char, vector<char>> adj;

public:
    void addEdge(char u, char v) {
        adj[u].push_back(v);
        adj[v].push_back(u);
    }

    void BFS(char start) {
        set<char> visited;
        queue<char> q;
        visited.insert(start);
        q.push(start);
        while (!q.empty()) {
            char cur = q.front(); q.pop();
            cout << cur << " ";
            for (char neighbor : adj[cur])
                if (visited.find(neighbor) == visited.end()) {
                    visited.insert(neighbor);
                    q.push(neighbor);
                }
        }
        cout << endl;
    }

    void DFS(char start) {
        set<char> visited;
        stack<char> s;
        s.push(start);
        while (!s.empty()) {
            char cur = s.top(); s.pop();
            if (visited.find(cur) != visited.end()) continue;
            visited.insert(cur);
            cout << cur << " ";
            for (char neighbor : adj[cur])
                if (visited.find(neighbor) == visited.end())
                    s.push(neighbor);
        }
        cout << endl;
    }
};

int main() {
    Graph g;
    g.addEdge('A', 'B');
    g.addEdge('A', 'C');
    g.addEdge('B', 'D');
    g.addEdge('B', 'E');
    g.addEdge('C', 'F');

    cout << "BFS from A: ";
    g.BFS('A');
    cout << "DFS from A: ";
    g.DFS('A');
    return 0;
}
import java.util.*;

class Graph {
    private Map<Character, List<Character>> adj = new HashMap<>();

    void addEdge(char u, char v) {
        adj.computeIfAbsent(u, k -> new ArrayList<>()).add(v);
        adj.computeIfAbsent(v, k -> new ArrayList<>()).add(u);
    }

    void BFS(char start) {
        Set<Character> visited = new HashSet<>();
        Queue<Character> q = new LinkedList<>();
        visited.add(start);
        q.offer(start);
        while (!q.isEmpty()) {
            char cur = q.poll();
            System.out.print(cur + " ");
            for (char n : adj.getOrDefault(cur, new ArrayList<>()))
                if (!visited.contains(n)) {
                    visited.add(n);
                    q.offer(n);
                }
        }
        System.out.println();
    }

    void DFS(char start) {
        Set<Character> visited = new HashSet<>();
        Stack<Character> s = new Stack<>();
        s.push(start);
        while (!s.isEmpty()) {
            char cur = s.pop();
            if (visited.contains(cur)) continue;
            visited.add(cur);
            System.out.print(cur + " ");
            for (char n : adj.getOrDefault(cur, new ArrayList<>()))
                if (!visited.contains(n))
                    s.push(n);
        }
        System.out.println();
    }
}

class Graph {
    constructor() {
        this.adj = new Map();
    }

    addEdge(u, v) {
        if (!this.adj.has(u)) this.adj.set(u, []);
        if (!this.adj.has(v)) this.adj.set(v, []);
        this.adj.get(u).push(v);
        this.adj.get(v).push(u);
    }

    BFS(start) {
        const visited = new Set();
        const queue = [start];
        visited.add(start);
        while (queue.length > 0) {
            const cur = queue.shift();
            process.stdout.write(cur + " ");
            for (const neighbor of (this.adj.get(cur) || []))
                if (!visited.has(neighbor)) {
                    visited.add(neighbor);
                    queue.push(neighbor);
                }
        }
        console.log();
    }

    DFS(start) {
        const visited = new Set();
        const stack = [start];
        while (stack.length > 0) {
            const cur = stack.pop();
            if (visited.has(cur)) continue;
            visited.add(cur);
            process.stdout.write(cur + " ");
            for (const neighbor of (this.adj.get(cur) || []))
                if (!visited.has(neighbor))
                    stack.push(neighbor);
        }
        console.log();
    }
}

<?php

class Graph {
    private array $adj = [];

    public function addEdge($u, $v) {
        $this->adj[$u][] = $v;
        $this->adj[$v][] = $u;
    }

    public function BFS($start) {
        $visited = [$start => true];
        $queue = [$start];
        while (!empty($queue)) {
            $cur = array_shift($queue);
            echo $cur . " ";
            foreach ($this->adj[$cur] ?? [] as $neighbor)
                if (!isset($visited[$neighbor])) {
                    $visited[$neighbor] = true;
                    $queue[] = $neighbor;
                }
        }
        echo "\n";
    }

    public function DFS($start) {
        $visited = [];
        $stack = [$start];
        while (!empty($stack)) {
            $cur = array_pop($stack);
            if (isset($visited[$cur])) continue;
            $visited[$cur] = true;
            echo $cur . " ";
            foreach ($this->adj[$cur] ?? [] as $neighbor)
                if (!isset($visited[$neighbor]))
                    $stack[] = $neighbor;
        }
        echo "\n";
    }
}
