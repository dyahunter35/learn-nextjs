#include <iostream>
#include <vector>
using namespace std;

bool isSafe(vector<int>& board, int row, int col) {
    for (int i = 0; i < col; i++) {
        if (board[i] == row ||
            abs(board[i] - row) == abs(i - col))
            return false;
    }
    return true;
}

void solve(vector<int>& board, int col, int n, vector<vector<int>>& solutions) {
    if (col == n) {
        solutions.push_back(board);
        return;
    }
    for (int row = 0; row < n; row++) {
        if (isSafe(board, row, col)) {
            board[col] = row;
            solve(board, col + 1, n, solutions);
        }
    }
}

void printBoard(vector<int>& board) {
    int n = board.size();
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++)
            cout << (board[j] == i ? "Q " : ". ");
        cout << endl;
    }
    cout << endl;
}

int main() {
    int n = 4;
    vector<vector<int>> solutions;
    vector<int> board(n, -1);
    solve(board, 0, n, solutions);
    cout << "Found " << solutions.size() << " solutions for " << n << "-Queens:" << endl;
    for (auto& sol : solutions) {
        printBoard(sol);

import java.util.*;

public class NQueens {
    static boolean isSafe(List<Integer> board, int row, int col) {
        for (int i = 0; i < col; i++) {
            if (board.get(i) == row ||
                Math.abs(board.get(i) - row) == Math.abs(i - col))
                return false;
        }
        return true;
    }

    static void solve(List<Integer> board, int col, int n, List<List<Integer>> solutions) {
        if (col == n) {
            solutions.add(new ArrayList<>(board));
            return;
        }
        for (int row = 0; row < n; row++) {
            if (isSafe(board, row, col)) {
                board.set(col, row);
                solve(board, col + 1, n, solutions);
            }
        }
    }

    static void printBoard(List<Integer> board) {
        int n = board.size();
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++)
                System.out.print(board.get(j) == i ? "Q " : ". ");
            System.out.println();
        }
        System.out.println();
    }

    public static void main(String[] args) {
        int n = 4;
        List<List<Integer>> solutions = new ArrayList<>();
        List<Integer> board = new ArrayList<>(Collections.nCopies(n, -1));
        solve(board, 0, n, solutions);
        System.out.println("Found " + solutions.size() + " solutions for " + n + "-Queens:");
        for (List<Integer> sol : solutions) printBoard(sol);
    }
}
