محول الأنظمة العددية — الكود البرمجي

شرح مبسط مع أمثلة بلغات متعددة. يمكنك نسخ الكود واستخدامه.

كيف تعمل: دوال التحويل بين الأنظمة العددية: decToBin (عشري ← ثنائي)، binToDec (ثنائي ← عشري)، decToHex (عشري ← سداسي عشري)، hexToDec (سداسي عشري ← عشري)، decToOct (عشري ← ثماني)، octToDec (ثماني ← عشري). تستخدم القسمة المتكررة للأساس المطلوب. 
#include <iostream>
#include <string>
#include <algorithm>
#include <sstream>
using namespace std;

string decToBin(int n) {
    if (n == 0) return "0";
    string res;
    while (n > 0) {
        res += (n % 2) + '0';
        n /= 2;
    }
    reverse(res.begin(), res.end());
    return res;
}

int binToDec(string s) {
    int res = 0;
    for (char c : s)
        res = res * 2 + (c - '0');
    return res;
}

string decToHex(int n) {
    if (n == 0) return "0";
    string res;
    while (n > 0) {
        int rem = n % 16;
        res += rem < 10 ? rem + '0' : rem - 10 + 'A';
        n /= 16;
    }
    reverse(res.begin(), res.end());
    return res;
}

int hexToDec(string s) {
    int res = 0;
    for (char c : s) {
        int val;
        if (c >= '0' && c <= '9') val = c - '0';
        else val = toupper(c) - 'A' + 10;
        res = res * 16 + val;
    }
    return res;
}

string decToOct(int n) {
    if (n == 0) return "0";
    string res;
    while (n > 0) {
        res += (n % 8) + '0';
        n /= 8;
    }
    reverse(res.begin(), res.end());
    return res;
}

int octToDec(string s) {
    int res = 0;
    for (char c : s)
        res = res * 8 + (c - '0');
    return res;
}

int main() {
    int n = 42;
    cout << "decToBin(" << n << ") = " << decToBin(n) << endl;
    cout << "binToDec(\"101010\") = " << binToDec("101010") << endl;
    cout << "decToHex(" << n << ") = " << decToHex(n) << endl;
    cout << "hexToDec(\"2A\") = " << hexToDec("2A") << endl;
    cout << "decToOct(" << n << ") = " << decToOct(n) << endl;
    cout << "octToDec(\"52\") = " << octToDec("52") << endl;
    return 0;
}
public class Converters {

    public static String decToBin(int n) {
        if (n == 0) return "0";
        StringBuilder sb = new StringBuilder();
        while (n > 0) {
            sb.append(n % 2);
            n /= 2;
        }
        return sb.reverse().toString();
    }

    public static int binToDec(String s) {
        int res = 0;
        for (int i = 0; i < s.length(); i++)
            res = res * 2 + (s.charAt(i) - '0');
        return res;
    }

    public static String decToHex(int n) {
        if (n == 0) return "0";
        StringBuilder sb = new StringBuilder();
        while (n > 0) {
            int rem = n % 16;
            sb.append(rem < 10 ? (char)(rem + '0') : (char)(rem - 10 + 'A'));
            n /= 16;
        }
        return sb.reverse().toString();
    }

    public static int hexToDec(String s) {
        int res = 0;
        s = s.toUpperCase();
        for (int i = 0; i < s.length(); i++) {
            char c = s.charAt(i);
            int val;
            if (c >= '0' && c <= '9') val = c - '0';
            else val = c - 'A' + 10;
            res = res * 16 + val;
        }
        return res;
    }

    public static String decToOct(int n) {
        if (n == 0) return "0";
        StringBuilder sb = new StringBuilder();
        while (n > 0) {
            sb.append(n % 8);
            n /= 8;
        }
        return sb.reverse().toString();
    }

    public static int octToDec(String s) {
        int res = 0;
        for (int i = 0; i < s.length(); i++)
            res = res * 8 + (s.charAt(i) - '0');
        return res;
    }

    public static void main(String[] args) {
        int n = 42;
        System.out.println("decToBin(" + n + ") = " + decToBin(n));
        System.out.println("binToDec(\"101010\") = " + binToDec("101010"));
        System.out.println("decToHex(" + n + ") = " + decToHex(n));
        System.out.println("hexToDec(\"2A\") = " + hexToDec("2A"));
        System.out.println("decToOct(" + n + ") = " + decToOct(n));
        System.out.println("octToDec(\"52\") = " + octToDec("52"));
    }
}
function decToBin(n) {
    if (n === 0) return "0";
    let res = "";
    while (n > 0) {
        res = (n % 2) + res;
        n = Math.floor(n / 2);
    }
    return res;
}

function binToDec(s) {
    let res = 0;
    for (const c of s)
        res = res * 2 + (c - '0');
    return res;
}

function decToHex(n) {
    if (n === 0) return "0";
    let res = "";
    while (n > 0) {
        const rem = n % 16;
        res = (rem < 10 ? rem : String.fromCharCode(rem - 10 + 65)) + res;
        n = Math.floor(n / 16);
    }
    return res;
}

function hexToDec(s) {
    let res = 0;
    for (const c of s.toUpperCase()) {
        const val = c >= '0' && c <= '9' ? c - '0' : c.charCodeAt(0) - 65 + 10;
        res = res * 16 + val;
    }
    return res;
}

function decToOct(n) {
    if (n === 0) return "0";
    let res = "";
    while (n > 0) {
        res = (n % 8) + res;
        n = Math.floor(n / 8);
    }
    return res;
}

function octToDec(s) {
    let res = 0;
    for (const c of s)
        res = res * 8 + (c - '0');
    return res;
}
