جداول الصواب المنطقية — الكود البرمجي

شرح مبسط مع أمثلة بلغات متعددة. يمكنك نسخ الكود واستخدامه.

كيف تعمل: دوال العمليات المنطقية الأساسية: AND, OR, NOT, NAND, NOR, XOR, XNOR. ودالة generateTruthTable(variables, expression) تولد جدول الصواب الكامل لأي تعبير منطقي.

bool XNOR(bool a, bool b) { return a == b; }

void generateTruthTable(vector<string> vars, string expr) {
    int n = vars.size();
    int rows = pow(2, n);
    for (string v : vars) cout << v << "\t";
    cout << expr << endl;
    for (int i = 0; i < rows; i++) {
        map<string, bool> values;
        for (int j = 0; j < n; j++)
            values[vars[j]] = (i >> (n - 1 - j)) & 1;
        for (int j = 0; j < n; j++)
            cout << values[vars[j]] << "\t";
        // Evaluate expression manually for this example
        // Here we show a simple hardcoded evaluation for "A AND B" or "A OR B"
        bool a = values["A"], b = values["B"];
        bool result = false;
        if (expr == "A AND B") result = AND(a, b);
        else if (expr == "A OR B") result = OR(a, b);
        else if (expr == "NOT A") result = NOT(a);
        else if (expr == "A NAND B") result = NAND(a, b);
        else if (expr == "A NOR B") result = NOR(a, b);
        else if (expr == "A XOR B") result = XOR(a, b);
        else if (expr == "A XNOR B") result = XNOR(a, b);
        cout << result << endl;
    }
}

import java.util.*;

public class TruthTables {

    public static boolean AND(boolean a, boolean b) { return a && b; }
    public static boolean OR(boolean a, boolean b) { return a || b; }
    public static boolean NOT(boolean a) { return !a; }
    public static boolean NAND(boolean a, boolean b) { return !(a && b); }
    public static boolean NOR(boolean a, boolean b) { return !(a || b); }
    public static boolean XOR(boolean a, boolean b) { return a != b; }
    public static boolean XNOR(boolean a, boolean b) { return a == b; }

    public static void generateTruthTable(List<String> vars, String expr) {
        int n = vars.size();
        int rows = (int) Math.pow(2, n);
        for (String v : vars) System.out.print(v + "\t");
        System.out.println(expr);
        for (int i = 0; i < rows; i++) {
            Map<String, Boolean> values = new HashMap<>();
            for (int j = 0; j < n; j++)
                values.put(vars.get(j), ((i >> (n - 1 - j)) & 1) == 1);
            for (int j = 0; j < n; j++)
                System.out.print((values.get(vars.get(j)) ? 1 : 0) + "\t");
            boolean a = values.get("A"), b = values.get("B");
            boolean result = false;
            switch (expr) {
                case "A AND B": result = AND(a, b); break;
                case "A OR B": result = OR(a, b); break;
                case "NOT A": result = NOT(a); break;
                case "A NAND B": result = NAND(a, b); break;
                case "A NOR B": result = NOR(a, b); break;
                case "A XOR B": result = XOR(a, b); break;
                case "A XNOR B": result = XNOR(a, b); break;
            }
            System.out.println(result ? 1 : 0);
        }
    }

    public static void main(String[] args) {
        System.out.println("Truth table for A AND B:");
        generateTruthTable(Arrays.asList("A", "B"), "A AND B");

        System.out.println("\nTruth table for A OR B:");
        generateTruthTable(Arrays.asList("A", "B"), "A OR B");

        System.out.println("\nTruth table for NOT A:");
        generateTruthTable(Arrays.asList("A"), "NOT A");

        System.out.println("\nBasic gates:");
        System.out.println("AND(1,0) = " + (AND(true, false) ? 1 : 0));
        System.out.println("OR(1,0) = " + (OR(true, false) ? 1 : 0));
        System.out.println("NOT(1) = " + (NOT(true) ? 1 : 0));
        System.out.println("NAND(1,0) = " + (NAND(true, false) ? 1 : 0));
        System.out.println("NOR(1,0) = " + (NOR(true, false) ? 1 : 0));
        System.out.println("XOR(1,0) = " + (XOR(true, false) ? 1 : 0));
        System.out.println("XNOR(1,0) = " + (XNOR(true, false) ? 1 : 0));
    }
}

            case "A XNOR B": result = XNOR(a, b); break;

}
