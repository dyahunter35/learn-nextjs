المكدس والطابور — الكود البرمجي

شرح مبسط لهياكل بيانات المكدس والطابور مع أمثلة برمجية بلغات متعددة. يمكنك نسخ الكود واستخدامه مباشرة.

كيف تعمل هياكل البيانات:
المكدس (Stack): هو هيكل بيانات يتبع مبدأ LIFO (Last In First Out) أي أن آخر عنصر تم إضافته هو أول عنصر يتم إزالته. العمليات الأساسية: push (إضافة إلى الأعلى)، pop (إزالة من الأعلى)، peek (مشاهدة العنصر العلوي).

الطابور (Queue): هو هيكل بيانات يتبع مبدأ FIFO (First In First Out) أي أن أول عنصر تم إضافته هو أول عنصر يتم إزالته. العمليات الأساسية: enqueue (إضافة إلى النهاية)، dequeue (إزالة من البداية)، peek (مشاهدة العنصر الأول). 
#include <iostream>
#include <vector>
using namespace std;

class Stack {
    vector<int> data;
public:
    void push(int val) { data.push_back(val); }

    int pop() {
        if (isEmpty()) throw runtime_error("Stack empty");
        int val = data.back();
        data.pop_back();
        return val;
    }

    int peek() {
        if (isEmpty()) throw runtime_error("Stack empty");
        return data.back();
    }

    bool isEmpty() { return data.empty(); }

    void display() {
        cout << "Stack (top to bottom): ";
        for (int i = data.size() - 1; i >= 0; i--) cout << data[i] << " ";
        cout << endl;
    }
};

class Queue {
    vector<int> data;
public:
    void enqueue(int val) { data.push_back(val); }

    int dequeue() {
        if (isEmpty()) throw runtime_error("Queue empty");
        int val = data.front();
        data.erase(data.begin());
        return val;
    }

    int peek() {
        if (isEmpty()) throw runtime_error("Queue empty");
        return data.front();
    }

    bool isEmpty() { return data.empty(); }

    void display() {
        cout << "Queue (front to back): ";
        for (int x : data) cout << x << " ";
        cout << endl;
    }
};

int main() {
    cout << "=== Stack Demo ===" << endl;
    Stack s;
    s.push(10); s.push(20); s.push(30);
    s.display();
    cout << "Popped: " << s.pop() << endl;
    cout << "Top: " << s.peek() << endl;
    s.display();

    cout << "\n=== Queue Demo ===" << endl;
    Queue q;
    q.enqueue(10); q.enqueue(20); q.enqueue(30);
    q.display();
    cout << "Dequeued: " << q.dequeue() << endl;
    cout << "Front: " << q.peek() << endl;
    q.display();

    return 0;
}
import java.util.LinkedList;

class Stack {
    private LinkedList<Integer> list = new LinkedList<>();

    public void push(int val) { list.addLast(val); }

    public int pop() {
        if (isEmpty()) throw new RuntimeException("Stack empty");
        return list.removeLast();
    }

    public int peek() {
        if (isEmpty()) throw new RuntimeException("Stack empty");
        return list.getLast();
    }

    public boolean isEmpty() { return list.isEmpty(); }

    public void display() {
        System.out.print("Stack (top to bottom): ");
        for (int i = list.size() - 1; i >= 0; i--) System.out.print(list.get(i) + " ");
        System.out.println();
    }
}

class Queue {
    private LinkedList<Integer> list = new LinkedList<>();

    public void enqueue(int val) { list.addLast(val); }

    public int dequeue() {
        if (isEmpty()) throw new RuntimeException("Queue empty");
        return list.removeFirst();
    }

    public int peek() {
        if (isEmpty()) throw new RuntimeException("Queue empty");
        return list.getFirst();
    }

    public boolean isEmpty() { return list.isEmpty(); }

    public void display() {
        System.out.print("Queue (front to back): ");
        for (int x : list) System.out.print(x + " ");
        System.out.println();
    }
}

class Stack {
    constructor() { this.data = []; }
    push(val) { this.data.push(val); }
    pop() {
        if (this.isEmpty()) throw new Error("Stack empty");
        return this.data.pop();
    }
    peek() {
        if (this.isEmpty()) throw new Error("Stack empty");
        return this.data[this.data.length - 1];
    }
    isEmpty() { return this.data.length === 0; }
    display() {
        console.log("Stack (top to bottom):", [...this.data].reverse().join(" "));
    }
}

class Queue {
    constructor() { this.data = []; }
    enqueue(val) { this.data.push(val); }
    dequeue() {
        if (this.isEmpty()) throw new Error("Queue empty");
        return this.data.shift();
    }
    peek() {
        if (this.isEmpty()) throw new Error("Queue empty");
        return this.data[0];
    }
    isEmpty() { return this.data.length === 0; }
    display() {
        console.log("Queue (front to back):", this.data.join(" "));
    }
}

<?php

class Stack {
    private $data = [];

    public function push($val) { array_push($this->data, $val); }

    public function pop() {
        if ($this->isEmpty()) throw new Exception("Stack empty");
        return array_pop($this->data);
    }

    public function peek() {
        if ($this->isEmpty()) throw new Exception("Stack empty");
        return end($this->data);
    }

    public function isEmpty() { return empty($this->data); }

    public function display() {
        $reversed = array_reverse($this->data);
        echo "Stack (top to bottom): " . implode(" ", $reversed) . "\n";
    }
}

class Queue {
    private $data = [];

    public function enqueue($val) { array_push($this->data, $val); }

    public function dequeue() {
        if ($this->isEmpty()) throw new Exception("Queue empty");
        return array_shift($this->data);
    }

    public function peek() {
        if ($this->isEmpty()) throw new Exception("Queue empty");
        return $this->data[0];
    }

    public function isEmpty() { return empty($this->data); }

    public function display() {
        echo "Queue (front to back): " . implode(" ", $this->data) . "\n";
    }
}

echo "=== Stack Demo ===\n";
$s = new Stack();

$s->display();
