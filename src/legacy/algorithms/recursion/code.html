العودية — الكود البرمجي

شرح مبسط مع أمثلة بلغات متعددة. يمكنك نسخ الكود واستخدامه.

كيف تعمل: العودية (Recursion) هي تقنية تستدعي فيها الدالة نفسها لحل مسألة بتقسيمها إلى مسائل فرعية أصغر. دالة factorial(n) تحسب مضروب العدد، ودالة fibonacci(n) تحسب حد متتالية فيبوناتشي. تم تضمين النسخة التكرارية للمقارنة. 
#include <iostream>
using namespace std;

// Recursive factorial
int factorialRec(int n) {
    if (n <= 1) return 1;
    return n * factorialRec(n - 1);
}

// Iterative factorial
int factorialIter(int n) {
    int result = 1;
    for (int i = 2; i <= n; i++)
        result *= i;
    return result;
}

// Recursive fibonacci
int fibRec(int n) {
    if (n <= 1) return n;
    return fibRec(n - 1) + fibRec(n - 2);
}

// Iterative fibonacci
int fibIter(int n) {
    if (n <= 1) return n;
    int a = 0, b = 1;
    for (int i = 2; i <= n; i++) {
        int c = a + b;
        a = b;
        b = c;
    }
    return b;
}

int main() {
    int n = 6;
    cout << "factorialRec(" << n << ") = " << factorialRec(n) << endl;
    cout << "factorialIter(" << n << ") = " << factorialIter(n) << endl;
    cout << "fibRec(" << n << ") = " << fibRec(n) << endl;
    cout << "fibIter(" << n << ") = " << fibIter(n) << endl;
    return 0;
}
