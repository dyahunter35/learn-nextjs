البرمجة الديناميكية — الكود البرمجي

شرح مبسط مع أمثلة بلغات متعددة. يمكنك نسخ الكود واستخدامه.

كيف تعمل: البرمجة الديناميكية (DP) تحل المسائل بتقسيمها إلى مسائل فرعية وتخزين نتائجها لتجنب إعادة الحساب. fibDP(n) تحسب فيبوناتشي تصاعدياً (bottom-up). minCoins(coins[], amount) تحسب أقل عدد من العملات لتكوين مبلغ معين باستخدام جدول DP. 
#include <iostream>
#include <vector>
#include <algorithm>
#include <climits>
using namespace std;

int fibDP(int n) {
    if (n <= 1) return n;
    vector<int> dp(n + 1);
    dp[0] = 0;
    dp[1] = 1;
    for (int i = 2; i <= n; i++)
        dp[i] = dp[i - 1] + dp[i - 2];
    return dp[n];
}

int minCoins(vector<int>& coins, int amount) {
    vector<int> dp(amount + 1, INT_MAX);
    dp[0] = 0;
    for (int i = 1; i <= amount; i++)
        for (int coin : coins)
            if (coin <= i && dp[i - coin] != INT_MAX)
                dp[i] = min(dp[i], 1 + dp[i - coin]);
    return dp[amount] == INT_MAX ? -1 : dp[amount];
}

int main() {
    int n = 10;
    cout << "fibDP(" << n << ") = " << fibDP(n) << endl;

    vector<int> coins = {1, 3, 4};
    int amount = 6;
    int result = minCoins(coins, amount);
    cout << "minCoins for amount " << amount << " = " << result << endl;
    return 0;
}

public class DP {

    public static int fibDP(int n) {
        if (n <= 1) return n;
        int[] dp = new int[n + 1];
        dp[0] = 0;
        dp[1] = 1;
        for (int i = 2; i <= n; i++)
            dp[i] = dp[i - 1] + dp[i - 2];
        return dp[n];
    }

    public static int minCoins(int[] coins, int amount) {
        int[] dp = new int[amount + 1];
        Arrays.fill(dp, Integer.MAX_VALUE);
        dp[0] = 0;
        for (int i = 1; i <= amount; i++)
            for (int coin : coins)
                if (coin <= i && dp[i - coin] != Integer.MAX_VALUE)
                    dp[i] = Math.min(dp[i], 1 + dp[i - coin]);
        return dp[amount] == Integer.MAX_VALUE ? -1 : dp[amount];
    }

    public static void main(String[] args) {
        int n = 10;
        System.out.println("fibDP(" + n + ") = " + fibDP(n));

        int[] coins = {1, 3, 4};
        int amount = 6;
        int result = minCoins(coins, amount);
        System.out.println("minCoins for amount " + amount + " = " + result);
    }
}
function fibDP(n) {
    if (n <= 1) return n;
    const dp = new Array(n + 1);
    dp[0] = 0;
    dp[1] = 1;
    for (let i = 2; i <= n; i++)
        dp[i] = dp[i - 1] + dp[i - 2];
    return dp[n];
}

function minCoins(coins, amount) {
    const dp = new Array(amount + 1).fill(Infinity);
    dp[0] = 0;
    for (let i = 1; i <= amount; i++)
        for (const coin of coins)
            if (coin <= i && dp[i - coin] !== Infinity)
                dp[i] = Math.min(dp[i], 1 + dp[i - coin]);
    return dp[amount] === Infinity ? -1 : dp[amount];
}
